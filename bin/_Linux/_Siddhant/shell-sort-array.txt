Write a shell script to sort an array of numbers using any sort method.
arr=($1 $2 $3 $4 $5)
echo "array in original order"
echo ${arr[*]}

for ((i = 0; i<5; i++))
do
        for (( j = 0; j < 5-i-1; j++))
        do
                if [ ${arr[j]} -gt ${arr[$((j+1))]} ]
                then
                        temp=${arr[j]}
                        arr[$j]=${arr[$(($j+1))]}
                        arr[$((j+1))]=$temp
                fi
        done
done

echo "Array in sorted order : "
echo ${arr[*]}
